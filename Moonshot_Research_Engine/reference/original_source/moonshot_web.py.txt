#!/usr/bin/env python3
"""
BITGET MOONSHOT SCANNER — WEB DASHBOARD
=======================================

A web UI on top of bitget_moonshot_scanner_v2.py.

Run it, open the printed URL in your browser, then:
  - set your account size and risk %,
  - scan specific coins or auto-discover small/early candidates,
  - watch results stream into a sortable table with score bars + tier badges,
  - click any coin to see a candlestick chart with the entry / stop / TP
    levels drawn on it, plus the full mechanical trade plan.

No API key required (same public Bitget + CoinGecko data the scanner uses).

Start (local only):
    pip install -r requirements_moonshot.txt
    python moonshot_web.py
Then open:  http://127.0.0.1:5000


ACCESS FROM ANOTHER DEVICE
--------------------------
1) Same Wi-Fi / LAN (easiest):
       set MOONSHOT_HOST=0.0.0.0
       python moonshot_web.py
   Then on the other device open:  http://<this-PC-LAN-IP>:5000

2) Public URL from anywhere (no router setup) via a tunnel:
       set MOONSHOT_HOST=0.0.0.0
       python moonshot_web.py
       cloudflared tunnel --url http://localhost:5000   # or ngrok http 5000

3) Protect it once it is reachable by others:
       set MOONSHOT_TOKEN=some-long-secret
   Then browse to:  http://<host>:5000/?token=some-long-secret

4) Production server (recommended when exposed): install waitress and it is
   used automatically. Force the dev server with MOONSHOT_DEV=1.
"""

import bitget_moonshot_scanner_v2 as scanner

app = Flask(__name__)

STATE_PATH = Path(scanner.DEFAULT_STATE_FILE)
CG_CACHE = Path(scanner.COINGECKO_LIST_CACHE)
WATCH_PATH = Path(".moonshot_watchlist.json")
ALERTS_PATH = Path(".moonshot_alerts.json")

# Optional shared access token. If set (via env or --token), every request must
# carry it as ?token=... , an X-Token header, or a Bearer Authorization header.
ACCESS_TOKEN = os.environ.get("MOONSHOT_TOKEN", "").strip()

# Alert channels (all optional; no-op if unset).
TELEGRAM_TOKEN = os.environ.get("MOONSHOT_TELEGRAM_TOKEN", "").strip()
TELEGRAM_CHAT = os.environ.get("MOONSHOT_TELEGRAM_CHAT", "").strip()
WEBHOOK_URL = os.environ.get("MOONSHOT_WEBHOOK", "").strip()

# Background auto-scan scheduler (optional). Enable by setting the interval > 0.
AUTOSCAN_INTERVAL_MIN = float(os.environ.get("MOONSHOT_AUTOSCAN_MIN", "0") or 0)
AUTOSCAN_UNIVERSE = int(os.environ.get("MOONSHOT_AUTOSCAN_N", "25") or 25)
AUTOSCAN_MIN_ASYM = int(os.environ.get("MOONSHOT_AUTOSCAN_MIN_ASYM", "60") or 60)


def _token_ok() -> bool:
    if not ACCESS_TOKEN:
        return True  # no token configured => open (fine for localhost/LAN)
    supplied = (
        request.args.get("token", "")
        or request.headers.get("X-Token", "")
        or request.headers.get("Authorization", "").replace("Bearer ", "").strip()
    )
    return bool(supplied) and secrets.compare_digest(supplied, ACCESS_TOKEN)


@app.before_request
def _gate():
    if request.path == "/" or request.path.startswith("/static"):
        return None
    if not _token_ok():
        return jsonify({"error": "unauthorized: missing or wrong token"}), 401


def _clean(obj: Any) -> Any:
    """Make results JSON-safe (NaN/inf -> None)."""
    if isinstance(obj, dict):
        return {k: _clean(v) for k, v in obj.items()}
    if isinstance(obj, (list, tuple)):
        return [_clean(v) for v in obj]
    if isinstance(obj, float):
        if math.isnan(obj) or math.isinf(obj):
            return None
        return obj
    if isinstance(obj, (np.floating,)):
        f = float(obj)
        return None if (math.isnan(f) or math.isinf(f)) else f
    if isinstance(obj, (np.integer,)):
        return int(obj)
    if isinstance(obj, (np.bool_,)):
        return bool(obj)
    return obj


# ---------------------------------------------------------------------------
# Alerts: Telegram + generic webhook, plus a persistent server-side event log
# ---------------------------------------------------------------------------

_alerts_lock = threading.Lock()


def _load_alerts() -> Dict[str, Any]:
    if not ALERTS_PATH.exists():
        return {"events": [], "seen": {}}
    try:
        d = json.loads(ALERTS_PATH.read_text(encoding="utf-8"))
        d.setdefault("events", [])
        d.setdefault("seen", {})
        return d
    except Exception:
        return {"events": [], "seen": {}}


def _save_alerts(d: Dict[str, Any]) -> None:
    try:
        d["events"] = d.get("events", [])[-200:]  # keep last 200 events
        ALERTS_PATH.write_text(json.dumps(d, ensure_ascii=False, indent=2), encoding="utf-8")
    except Exception:
        pass


def send_telegram(text: str) -> bool:
    if not (TELEGRAM_TOKEN and TELEGRAM_CHAT):
        return False
    try:
        requests.post(
            f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage",
            json={"chat_id": TELEGRAM_CHAT, "text": text, "parse_mode": "HTML",
                  "disable_web_page_preview": True},
            timeout=10,
        )
        return True
    except Exception:
        return False


def send_webhook(payload: Dict[str, Any]) -> bool:
    if not WEBHOOK_URL:
        return False
    try:
        requests.post(WEBHOOK_URL, json=payload, timeout=10)
        return True
    except Exception:
        return False


def record_alert(kind: str, symbol: str, detail: Dict[str, Any]) -> None:
    """
    Record a de-duplicated alert event and fan it out to configured channels.
    De-dup key = kind+symbol; the same breakout won't spam every check.
    """
    with _alerts_lock:
        d = _load_alerts()
        dedup_key = f"{kind}:{symbol}"
        last = d["seen"].get(dedup_key, 0)
        now = time.time()
        # Suppress repeats within 6 hours.
        if now - last < 6 * 3600:
            return
        d["seen"][dedup_key] = now
        ev = {
            "ts": scanner.utc_now_iso(),
            "kind": kind,
            "symbol": symbol,
            **detail,
        }
        d["events"].append(ev)
        _save_alerts(d)

    price = detail.get("price")
    trig = detail.get("trigger")
    if kind == "BREAKOUT":
        msg = (f"🚀 <b>BREAKOUT</b> {symbol}\nprice {price} broke trigger {trig}\n"
               f"tier {detail.get('tier','-')} · verdict {detail.get('verdict','-')}")
    elif kind == "INVALIDATED":
        msg = f"🛑 <b>INVALIDATED</b> {symbol}\nprice {price} broke stop {detail.get('stop')}"
    else:
        msg = f"{kind} {symbol}"
    send_telegram(msg)
    send_webhook({"kind": kind, "symbol": symbol, **detail})


def check_watch_and_alert() -> List[Dict[str, Any]]:
    """
    Re-price every watched coin, update status, and fire alerts on new
    BREAKOUT / INVALIDATED transitions. Returns the current status rows.
    """
    d = _load_watch()
    out = []
    for sym, it in d.get("items", {}).items():
        row = {"symbol": sym, "trigger": it.get("trigger"), "stop": it.get("stop"),
               "tier": it.get("tier"), "phase": it.get("phase"), "verdict": it.get("verdict"),
               "price": None, "status": "unknown"}
        try:
            t = scanner.fetch_ticker(sym, "SPOT")
            price = scanner.sf(t.get("lastPrice"), scanner.sf(t.get("close")))
            if price == price:  # not NaN
                row["price"] = float(price)
                trig = scanner.sf(it.get("trigger"))
                stop = scanner.sf(it.get("stop"))
                if stop and price <= stop:
                    row["status"] = "INVALIDATED"
                    record_alert("INVALIDATED", sym, {"price": row["price"], "stop": float(stop)})
                elif trig and price >= trig:
                    row["status"] = "BREAKOUT"
                    record_alert("BREAKOUT", sym, {"price": row["price"], "trigger": float(trig),
                                                   "tier": it.get("tier"), "verdict": it.get("verdict")})
                elif trig:
                    row["status"] = "WAITING"
                    row["to_trigger_pct"] = (trig / price - 1.0) * 100.0 if price > 0 else None
        except Exception as e:
            row["status"] = "error"
            row["error"] = str(e)
        out.append(_clean(row))
    return out


# ---------------------------------------------------------------------------
# Background auto-scan scheduler (optional)
# ---------------------------------------------------------------------------
_scheduler_started = False


def _autoscan_loop():
    """
    Periodically re-check the watchlist (so breakouts alert even with no browser
    open) and, if a universe size is set, refresh a moonshot scan whose top
    picks get auto-added to the watchlist. Runs in a daemon thread.
    """
    # Small initial delay so the web server binds first.
    time.sleep(20)
    while True:
        try:
            check_watch_and_alert()
        except Exception:
            pass

        # Self-improvement: grade matured predictions and re-tune weights.
        try:
            run_learning_cycle()
        except Exception:
            pass

        if AUTOSCAN_UNIVERSE > 0:
            try:
                state = scanner.load_state(STATE_PATH)
                btc = scanner.btc_regime()
                syms = scanner.discover_candidates(
                    AUTOSCAN_UNIVERSE, 500_000, max_turnover_usdt=50_000_000, moonshot_mode=True
                )
                w = _load_watch()
                for sym in syms:
                    try:
                        r = scanner.analyze_symbol(sym, state, btc, CG_CACHE, journal=True)
                        if (r.get("asymmetry_score") or 0) >= AUTOSCAN_MIN_ASYM and \
                           r.get("phase") in ("COILED_EARLY", "EARLY_BREAKOUT", "BASING"):
                            w.setdefault("items", {})[sym] = {
                                "symbol": sym, "trigger": r.get("trigger_level"),
                                "entry_low": r.get("entry_low"), "entry_high": r.get("entry_high"),
                                "stop": r.get("stop"), "tp1": r.get("tp1"),
                                "phase": r.get("phase"), "tier": r.get("moonshot_tier"),
                                "verdict": r.get("verdict"), "added_utc": scanner.utc_now_iso(),
                                "auto": True,
                            }
                    except Exception:
                        pass
                    time.sleep(scanner.REQUEST_SLEEP)
                _save_watch(w)
                scanner.save_state(STATE_PATH, state)
            except Exception:
                pass

        time.sleep(max(60.0, AUTOSCAN_INTERVAL_MIN * 60.0))


def _start_scheduler_once():
    global _scheduler_started
    if _scheduler_started:
        return
    if AUTOSCAN_INTERVAL_MIN <= 0:
        return
    _scheduler_started = True
    threading.Thread(target=_autoscan_loop, daemon=True).start()

@app.route("/api/candles")
def api_candles() -> Response:
    """OHLC candles for the chart. e.g. /api/candles?symbol=SUIUSDT&interval=4H"""
    symbol = request.args.get("symbol", "BTCUSDT")
    interval = request.args.get("interval", "4H")
    try:
        limit = int(request.args.get("limit", "200"))
    except ValueError:
        limit = 200

    try:
        df = scanner.fetch_candles(symbol, interval, min(limit, 1000), "SPOT")
    except Exception as e:
        return jsonify({"error": str(e)}), 400

    candles = [
        {
            "time": int(row.timestamp // 1000),
            "open": float(row.open),
            "high": float(row.high),
            "low": float(row.low),
            "close": float(row.close),
            "volume": float(row.volume),
        }
        for row in df.itertuples(index=False)
    ]
    return jsonify({"symbol": symbol, "interval": interval, "candles": candles})


@app.route("/api/scan")
def api_scan() -> Response:
    """Stream scan results as newline-delimited JSON (one object per line)."""
    symbols_param = request.args.get("symbols", "").strip()
    auto = int(request.args.get("auto", "0") or 0)
    min_turnover = float(request.args.get("min_turnover", "1000000") or 1000000)
    max_turnover = float(request.args.get("max_turnover", "0") or 0)
    account = float(request.args.get("account", "1000") or 1000)
    risk = float(request.args.get("risk", "1.0") or 1.0)
    min_asym = int(request.args.get("min_asymmetry", "0") or 0)
    with_fund = request.args.get("fundamentals", "1") != "0"
    with_oc = request.args.get("onchain", "1") != "0"

    @stream_with_context
    def generate():
        state = scanner.load_state(STATE_PATH)

        btc = scanner.btc_regime()
        yield json.dumps({"type": "regime", "regime": _clean(btc)}) + "\n"

        if symbols_param:
            symbols = [scanner.symbol_clean(s) for s in symbols_param.split(",") if s.strip()]
        elif auto > 0:
            symbols = scanner.discover_candidates(
                auto, min_turnover,
                max_turnover_usdt=max_turnover,
                moonshot_mode=True,
            )
        else:
            symbols = scanner.DEFAULT_SYMBOLS

        yield json.dumps({"type": "start", "symbols": symbols, "count": len(symbols)}) + "\n"

        for idx, sym in enumerate(symbols, start=1):
            try:
                r = scanner.analyze_symbol(
                    sym, state, btc, CG_CACHE,
                    account_usd=account,
                    risk_pct=risk,
                    with_fundamentals=with_fund,
                    with_onchain=with_oc,
                    journal=True,
                )
                asym = r.get("asymmetry_score") or 0
                if min_asym and asym < min_asym:
                    yield json.dumps({"type": "skip", "symbol": sym, "index": idx}) + "\n"
                    continue
                yield json.dumps({"type": "result", "index": idx, "data": _clean(r)}) + "\n"
            except Exception as e:
                yield json.dumps({"type": "error", "symbol": sym, "index": idx, "message": str(e)}) + "\n"

        try:
            scanner.save_state(STATE_PATH, state)
        except Exception:
            pass

        yield json.dumps({"type": "done"}) + "\n"

    return Response(generate(), mimetype="application/x-ndjson")


# ---------------------------------------------------------------------------
# Watchlist + breakout alerts
# ---------------------------------------------------------------------------
# The whole point of catching a COILED_EARLY / BASING coin is to act WHEN it
# actually breaks out. This layer lets you save a coin with its breakout
# trigger level, then re-check live prices and flag any that have crossed it.
# Storage is a small JSON file. On Render's ephemeral disk it resets on
# redeploy/sleep, which is fine for a personal watchlist; the front-end also
# keeps a copy in localStorage as a backup.


def _load_watch() -> Dict[str, Any]:
    if not WATCH_PATH.exists():
        return {"items": {}}
    try:
        d = json.loads(WATCH_PATH.read_text(encoding="utf-8"))
        d.setdefault("items", {})
        return d
    except Exception:
        return {"items": {}}


def _save_watch(d: Dict[str, Any]) -> None:
    try:
        WATCH_PATH.write_text(json.dumps(d, ensure_ascii=False, indent=2), encoding="utf-8")
    except Exception:
        pass


@app.route("/api/watch", methods=["GET", "POST", "DELETE"])
def api_watch() -> Response:
    d = _load_watch()

    if request.method == "GET":
        return jsonify({"items": list(d["items"].values())})

    body = request.get_json(silent=True) or {}
    sym = scanner.symbol_clean(str(body.get("symbol", "")))
    if not sym:
        return jsonify({"error": "symbol required"}), 400

    if request.method == "DELETE":
        d["items"].pop(sym, None)
        _save_watch(d)
        return jsonify({"ok": True, "items": list(d["items"].values())})

    # POST: add / update a watch entry with its breakout trigger.
    d["items"][sym] = {
        "symbol": sym,
        "trigger": float(body.get("trigger", 0) or 0),
        "entry_low": float(body.get("entry_low", 0) or 0),
        "entry_high": float(body.get("entry_high", 0) or 0),
        "stop": float(body.get("stop", 0) or 0),
        "tp1": float(body.get("tp1", 0) or 0),
        "phase": body.get("phase"),
        "tier": body.get("moonshot_tier"),
        "verdict": body.get("verdict"),
        "added_utc": scanner.utc_now_iso(),
    }
    _save_watch(d)
    return jsonify({"ok": True, "items": list(d["items"].values())})


@app.route("/api/watch/check")
def api_watch_check() -> Response:
    """
    Re-price every watched coin, update status, and fire alerts on NEW
    breakout / invalidation transitions (Telegram + webhook if configured).
    This is what turns 'pre-move' picks into actionable alerts.
    """
    return jsonify({"items": check_watch_and_alert()})


@app.route("/api/alerts")
def api_alerts() -> Response:
    """Return the recent server-side alert event feed (newest first)."""
    d = _load_alerts()
    return jsonify({
        "events": list(reversed(d.get("events", [])))[:60],
        "channels": {
            "telegram": bool(TELEGRAM_TOKEN and TELEGRAM_CHAT),
            "webhook": bool(WEBHOOK_URL),
        },
        "autoscan_min": AUTOSCAN_INTERVAL_MIN,
    })


@app.route("/api/discover")
def api_discover() -> Response:
    """
    Discovery feeds for finding genuinely early names:
      kind=new       -> recently listed Bitget spot pairs (fresh listings)
      kind=trending  -> DexScreener boosted/trending on-chain tokens
    """
    kind = request.args.get("kind", "new")
    try:
        if kind == "trending":
            return jsonify({"kind": kind, "items": _clean(scanner.dex_trending_tokens(30))})
        max_age = float(request.args.get("max_age_days", "30") or 30)
        return jsonify({"kind": "new", "items": _clean(scanner.new_listing_radar(max_age, 40))})
    except Exception as e:
        return jsonify({"error": str(e), "items": []}), 500


# ---------------------------------------------------------------------------
# Learning loop endpoints
# ---------------------------------------------------------------------------

def _spot_price(sym: str) -> float:
    t = scanner.fetch_ticker(sym, "SPOT")
    return scanner.sf(t.get("lastPrice"), scanner.sf(t.get("close")))


def run_learning_cycle() -> Dict[str, Any]:
    """Grade matured predictions against live prices, then re-tune weights."""
    counts = scanner.evaluate_predictions(_spot_price)
    stats = scanner.learn_from_outcomes()
    stats["graded_now"] = counts
    return stats


@app.route("/api/session")
def api_session() -> Response:
    """Current Yerevan-time market session + favorability."""
    return jsonify(_clean(scanner.session_profile()))


@app.route("/api/market")
def api_market() -> Response:
    """
    BTC + ETH market context: price, 24h, trend read. Alts largely follow these,
    so a bearish BTC/ETH is a reason to size down or stay flat on alt longs.
    """
    out = {}
    for sym in ("BTCUSDT", "ETHUSDT"):
        try:
            df = scanner.fetch_candles(sym, "4H", 200, "SPOT")
            snap = scanner.snapshot(df)
            t = scanner.fetch_ticker(sym, "SPOT")
            out[sym] = _clean({
                "price": snap["close"],
                "change_24h_pct": scanner.sf(t.get("price24hPcnt"), 0.0) * 100,
                "rsi_4h": snap["rsi"],
                "trend_4h": snap["trend_score"],
                "ema20": snap["ema20"], "ema50": snap["ema50"], "ema200": snap["ema200"],
                "above_ema200": bool(snap["close"] > snap["ema200"]),
            })
        except Exception as e:
            out[sym] = {"error": str(e)}
    try:
        out["regime"] = _clean(scanner.btc_regime())
    except Exception:
        out["regime"] = {}
    return jsonify(out)


@app.route("/api/learn/stats")
def api_learn_stats() -> Response:
    """
    The scanner's self-improvement report: hit rate, per-factor edge, current
    learned weights, sample size, and the recent prediction journal.
    """
    d = scanner.load_learn()
    preds = d.get("predictions", [])
    recent = list(reversed(preds))[:40]
    return jsonify({
        "weights": d.get("weights", {}),
        "stats": d.get("stats", {}),
        "predictions": _clean(recent),
        "total_predictions": len(preds),
        "min_samples": scanner.LEARN_MIN_SAMPLES,
        "horizon_hours": scanner.LEARN_HORIZON_HOURS,
        "hit_threshold_pct": scanner.LEARN_HIT_PCT,
    })


@app.route("/api/learn/evaluate", methods=["POST", "GET"])
def api_learn_evaluate() -> Response:
    """Manually trigger an evaluation + learning cycle (also runs on schedule)."""
    try:
        stats = run_learning_cycle()
        return jsonify({"ok": True, "stats": _clean(stats)})
    except Exception as e:
        return jsonify({"ok": False, "error": str(e)}), 500

function fmtPrice(x){
  if(x==null||!isFinite(x)) return "-";
  if(x>=1000) return x.toLocaleString(undefined,{maximumFractionDigits:2});
  if(x>=1) return x.toFixed(4);
  if(x>=0.01) return x.toFixed(6);
  return x.toExponential(2);
}
function fmtUsd(x){
  if(x==null||!isFinite(x)) return "-";
  const a=Math.abs(x);
  if(a>=1e9) return "$"+(x/1e9).toFixed(2)+"B";
  if(a>=1e6) return "$"+(x/1e6).toFixed(2)+"M";
  if(a>=1e3) return "$"+(x/1e3).toFixed(1)+"K";
  return "$"+x.toFixed(0);
}
function pct(x){ if(x==null||!isFinite(x)) return "-"; return (x>=0?"+":"")+x.toFixed(2)+"%"; }

function verdictClass(v){
  if(!v) return "neutral";
  if(v.includes("STRONG")) return "strong";

function factorBreakdown(r){
  let fs={}, fw={};
  try{ fs = JSON.parse(r.factor_scores||"{}"); }catch(e){ return ""; }
  try{ fw = JSON.parse(r.factor_weights||"{}"); }catch(e){ fw={}; }
  const rows = Object.entries(fs).map(([k,v])=>{
    const val = Math.max(-100,Math.min(100,v||0));
    const up = val>=0;
    const w = Math.abs(val)/2; // half-width max 50%
    const left = up? 50 : 50-w;
    const wt = fw[k]!=null? fw[k] : 1;
    const wtag = Math.abs(wt-1)>0.03 ? `<span style="color:${wt>1?'#7ff0a8':'#ff9fae'};font-size:10px">×${wt.toFixed(2)}</span>` : "";
    return `<div class="factorrow"><span class="fl">${k} ${wtag}</span>
      <span class="factortrack"><span class="mid"></span>
        <span class="factorfill ${up?'up':'dn'}" style="left:${left}%;width:${w}%"></span></span>
      <span class="fv ${up?'pos':'neg'}">${val>0?"+":""}${val.toFixed(0)}</span></div>`;
  }).join("");
  return `<div class="factors"><div class="muted" style="font-size:11px;text-transform:uppercase;letter-spacing:.5px;margin-bottom:4px">Factor confluence (each &minus;100&hellip;+100) &middot; ×weight = learned</div>${rows}</div>`;
}

async function scan(){
  results=[]; selected=null; done=0; total=0; renderTable();
  $("detail").innerHTML='<div class="empty"><div class="big">&#8987;</div>Scanning&hellip; results appear here as coins finish.</div>';
  $("dhint").textContent="no coin selected";
  $("progbar").classList.add("show"); $("progbar").firstElementChild.style.width="0%";
  const q = new URLSearchParams({
    symbols:$("symbols").value.trim(), auto:$("auto").value,
    min_turnover:$("minTurnover").value, max_turnover:$("maxTurnover").value,
    min_asymmetry:$("minAsym").value, account:$("account").value,
    risk:$("risk").value, fundamentals:$("fund").value, onchain:$("onchain").value,
  });
  $("scanBtn").disabled=true; $("status").textContent="Starting&hellip;";
  document.body.classList.add("scanning");
  try{
    const resp = await fetch(withToken("/api/scan?"+q.toString()));
    if(resp.status===401){ $("status").textContent="Unauthorized \u2014 add ?token=YOUR_TOKEN to the URL"; $("scanBtn").disabled=false; $("progbar").classList.remove("show"); return; }
    const reader = resp.body.getReader(); const dec=new TextDecoder(); let buf="";
    while(true){
      const {value,done:rd} = await reader.read(); if(rd) break;
      buf += dec.decode(value,{stream:true});
      let nl;
      while((nl=buf.indexOf("\n"))>=0){
        const line=buf.slice(0,nl).trim(); buf=buf.slice(nl+1); if(!line) continue;
        const msg=JSON.parse(line);
        if(msg.type==="regime"){ setRegime(msg.regime); }
        else if(msg.type==="start"){ total=msg.count; $("status").textContent=`Scanning ${total} coins\u2026`; }
        else if(msg.type==="result"){ done++; results.push(msg.data); renderTable(); tick(); }
        else if(msg.type==="skip"){ done++; tick(msg.symbol+" filtered out"); }
        else if(msg.type==="error"){ done++; tick("skipped "+msg.symbol); }
        else if(msg.type==="done"){ $("status").textContent=`Done \u2014 ${results.length} coins ranked.`; $("progbar").firstElementChild.style.width="100%";
          setTimeout(()=>$("progbar").classList.remove("show"),800);
          if(results.length && !selected) selectCoin(results[0].symbol); }
      }
    }
  }catch(e){ $("status").textContent="Error: "+e.message; $("progbar").classList.remove("show"); }
  $("scanBtn").disabled=false;
  document.body.classList.remove("scanning");
}
function tick(extra){
  const p = total? Math.round(done/total*100):0;
  $("progbar").firstElementChild.style.width=p+"%";
  $("status").textContent=`${done}/${total} analyzed \u00b7 ${results.length} shown${extra?" \u00b7 "+extra:""}`;
}

def _lan_ip() -> str:
    """Best-effort local network IP (the 192.168.x.x style address)."""
    import socket
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try:
        s.connect(("8.8.8.8", 80))  # no packets sent; just picks the route
        return s.getsockname()[0]
    except Exception:
        return "127.0.0.1"
    finally:
        s.close()
